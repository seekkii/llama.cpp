cmake_minimum_required(VERSION 3.14)
project(llama VERSION 1.0 LANGUAGES CXX)


set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT DEFINED PYTHON_EXECUTABLE)
    set(PYTHON_EXECUTABLE "/mnt/data/youtube-api-test/venv/bin/python" CACHE FILEPATH "Python interpreter")
endif()
find_package(Threads REQUIRED)

add_subdirectory(pybind11)

pybind11_add_module(_llama
    llama_python_bind.cpp
    llama_main.cpp
)
target_link_libraries(_llama
    PRIVATE
        llama
        common
        Threads::Threads
)

set_target_properties(_llama PROPERTIES
  # where to place the built shared‐object
  LIBRARY_OUTPUT_DIRECTORY "/mnt/data/youtube-api-test/"
)

set(LLAMA_PY_SRC
    llama_python_bind.cpp
)

target_include_directories(_llama PRIVATE
    $<BUILD_INTERFACE:${CMAKE_SOURCE_DIR}/include>
)
